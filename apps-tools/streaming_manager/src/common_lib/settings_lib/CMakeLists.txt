add_library(${PROJECT_NAME} ${INC_ALL})

target_include_directories(${PROJECT_NAME}
    PUBLIC  ${PROJECT_SOURCE_DIR}
            ${COMMON_LIB_DIR}
            ${json_SOURCE_DIR}/include)

list(APPEND headers
            ${PROJECT_SOURCE_DIR}/enums.h
            ${PROJECT_SOURCE_DIR}/dac_settings_client.h
            ${PROJECT_SOURCE_DIR}/stream_settings.h
        )

    list(APPEND src
            ${PROJECT_SOURCE_DIR}/stream_settings.cpp
        )

target_sources(${PROJECT_NAME} PRIVATE ${src})

if(SETTINGS_ENABLE_SANITIZERS)
    set(_settings_sanitizer_flags -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer -g)
    target_compile_options(${PROJECT_NAME} PUBLIC ${_settings_sanitizer_flags})
    target_link_options(${PROJECT_NAME} PUBLIC ${_settings_sanitizer_flags})
endif()

if (NOT CMAKE_CROSSCOMPILING)
    if(BUILD_UNIT_TESTS)
        enable_testing()
        add_subdirectory(tests)

        ## Runs the suite as part of the build, immediately after the library and
        ## the test binary are up to date, so a regression fails the build instead
        ## of waiting for someone to remember `ctest`.
        add_custom_target(rp_settings_lib_run_tests ALL
            ## Run from this library's own binary directory: enable_testing() above
            ## is scoped to this directory, so the test list generated by
            ## gtest_discover_tests lives here and not in the top-level build dir.
            COMMAND ${CMAKE_CTEST_COMMAND} --output-on-failure
            WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
            COMMENT "Running settings_lib unit tests (ctest) - the build fails if any test fails"
            VERBATIM)
        add_dependencies(rp_settings_lib_run_tests settings_lib_tests)
    endif()
endif()

        "${CMAKE_BINARY_DIR}/bin/include/${PROJECT_NAME}"
    COMMENT "Copying ${PROJECT_NAME} public headers to ${CMAKE_BINARY_DIR}/include/${PROJECT_NAME}"
    VERBATIM)
